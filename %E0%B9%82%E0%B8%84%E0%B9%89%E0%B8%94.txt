    function calculateInitialPoints(dataset) {
      dataset.members.forEach(member => {
        const memberDeposits =
          dataset.deposits.filter(
            deposit => deposit.memberId === member.id
          );
        const totalPoints =
          memberDeposits.reduce(
            (sum, deposit) => sum + deposit.points,
            0
          );
        const totalRedeemed =
          dataset.redemptions
          .filter(item => item.memberId === member.id)
          .reduce((sum, item) => sum + item.cost, 0);
        member.points = totalPoints - totalRedeemed;
      });
    }

    function confirmReward() {
      if (!selectedRewardId) return;
      const reward =
        data.rewards.find(
          item => item.id === selectedRewardId
        );
      const member =
        getMember(currentMemberId);
      if (!reward || !member) return;
      if (member.points < reward.cost) {
        showToast("แต้มไม่เพียงพอ");
        closeRewardModal();
        return;
      }
      member.points -= reward.cost;
      data.redemptions.push({
        id: Date.now(),
        memberId: member.id,
        rewardId: reward.id,
        rewardName: reward.name,
        icon: reward.icon,
        cost: reward.cost,
        date: getTodayThai()
      });
      saveData();
      closeRewardModal();
      renderUser();
      showToast(
        `แลก${reward.name}สำเร็จ`
      );
    }

    function calculateStaffPoints() {
      const type =
        document.getElementById("staff-waste").value;
      const weight =
        Number(
          document.getElementById("staff-weight").value
        );
      const rate =
        Number(data.rates[type] || 0);
      const points =
        weight > 0 ?
        weight * rate :
        0;
      document.getElementById(
          "staff-calculated-points"
        ).textContent =
        formatNumber(points);
    }

    function saveDeposit() {
      const memberId =
        Number(
          document.getElementById("staff-member").value
        );
      const type =
        document.getElementById("staff-waste").value;
      const weight =
        Number(
          document.getElementById("staff-weight").value
        );
      if (!memberId || !type || weight <= 0) {
        showToast(
          "กรุณากรอกข้อมูลการฝากขยะให้ครบถ้วน"
        );
        return;
      }
      const points =
        weight * data.rates[type];
      const member =
        getMember(memberId);
      member.points += points;
      data.deposits.push({
        id: Date.now(),
        memberId,
        date: getTodayThai(),
        type,
        weight,
        points,
        collector: currentStaffName()
      });
      saveData();
      document.getElementById(
        "staff-weight"
      ).value = "";
      document.getElementById(
        "staff-calculated-points"
      ).textContent = "0";
      renderStaff();
      showToast(
        `บันทึกสำเร็จ ห้องได้รับ ${formatNumber(points)} แต้ม`
      );
    }

    function registerRoom() {
      const roomName =
        document.getElementById("reg-room-name").value.trim();
      const studentId =
        document.getElementById("reg-room-studentid").value.trim();
      const studentName =
        document.getElementById("reg-room-studentname").value.trim();

      if (!roomName || !studentId || !studentName) {
        showToast("กรุณากรอกข้อมูลห้องให้ครบถ้วน");
        return;
      }
      if (data.rooms.some(r => r.room === roomName)) {
        showToast(`มีห้อง ${roomName} อยู่ในระบบแล้ว`);
        return;
      }

      const newId =
        data.rooms.length ?
        Math.max(...data.rooms.map(r => r.id)) + 1 :
        1;

      data.rooms.push({ id: newId, room: roomName, studentId, studentName });
      data.members.push({ id: newId, points: 0 });
      saveData();

      document.getElementById("reg-room-name").value = "";
      document.getElementById("reg-room-studentid").value = "";
      document.getElementById("reg-room-studentname").value = "";

      populateLoginRoomSelector();
      renderStaff();
      showToast(`ลงทะเบียนห้อง ${roomName} สำเร็จ`);
    }

    function clearRoomHistory(roomId) {
      const room = getRoomInfo(roomId);
      if (!room) return;
      const ok = confirm(
        `ล้างประวัติการฝากขยะและรีเซ็ตแต้มของห้อง ${room.room} ใช่หรือไม่?`
      );
      if (!ok) return;

      data.deposits = data.deposits.filter(d => d.memberId !== roomId);
      data.redemptions = data.redemptions.filter(r => r.memberId !== roomId);
      const member = getMember(roomId);
      if (member) member.points = 0;
      saveData();
      renderStaff();
      showToast(`ล้างประวัติห้อง ${room.room} เรียบร้อยแล้ว`);
    }

    function deleteRoom(roomId) {
      const room = getRoomInfo(roomId);
      if (!room) return;
      const ok = confirm(
        `ต้องการลบห้อง ${room.room} ออกจากระบบพร้อมประวัติทั้งหมดใช่หรือไม่?\nใช้เมื่อนักเรียนจบการศึกษา (ม.3 / ม.6)`
      );
      if (!ok) return;

      data.deposits = data.deposits.filter(d => d.memberId !== roomId);
      data.redemptions = data.redemptions.filter(r => r.memberId !== roomId);
      data.rooms = data.rooms.filter(r => r.id !== roomId);
      data.members = data.members.filter(m => m.id !== roomId);
      saveData();

      populateLoginRoomSelector();
      renderStaff();
      showToast(`ลบห้อง ${room.room} ออกจากระบบแล้ว`);
    }

    function deleteStaffAccount(staffId) {
      if (data.staffAccounts.length <= 1) {
        showToast("ต้องมีเจ้าหน้าที่อย่างน้อย 1 คนในระบบ");
        return;
      }
      const staff = data.staffAccounts.find(s => s.id === staffId);

      const ok = confirm(`ต้องการลบบัญชีเจ้าหน้าที่ ${staff.name} ใช่หรือไม่?`);
      if (!ok) return;

      data.staffAccounts = data.staffAccounts.filter(s => s.id !== staffId);
      saveData();
      renderStaff();
      showToast(`ลบบัญชีเจ้าหน้าที่ ${staff.name} แล้ว`);
    }
